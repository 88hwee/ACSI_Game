        function shuffleArray(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
            return array;
        }

            shuffledQuestions = shuffleArray([...questionsData]);

        function updateTimerDisplay() {
            const mins = Math.floor(timeLeft / 60).toString().padStart(2, '0');

        function setupCurrentQuestion() {
            if (currentQuestionIndex >= shuffledQuestions.length) {
                endGame("Mission Complete!", "Fantastic! You have mastered all revision questions!");
                return;
            }

            const qObj = shuffledQuestions[currentQuestionIndex];
            document.getElementById('questionText').innerText = qObj.question;
            
            let randomizedOptions = shuffleArray([...qObj.options]);
            
            aliens = [];
            const spacing = canvas.width / (randomizedOptions.length + 1);

            randomizedOptions.forEach((opt, idx) => {
                const alienX = spacing * (idx + 1) - 60;
                const alienY = 70 + (idx % 2) * 60; // staggered height for laptop view
                
                aliens.push({
                    x: alienX,
                    y: alienY,
                    width: 120,
                    height: 55,
                    text: opt,
                    isCorrect: (opt === qObj.correctAnswer),
                    speedX: (idx % 2 === 0 ? 1 : -1) * 1.2
                });
            });
        }

        function selectAnswer(selected, correct) {
            if (!gameRunning) return;

            if (selected === correct) {
                // Correct answer
                score += 100;
                answeredCount++;
                currentQuestionIndex++;
                createExplosion(player.x + 25, player.y, '#10b981');
                updateUI();
                setupCurrentQuestion();
            } else {
                // Incorrect answer: repeat question rule
                lives--;
                updateUI();
                if (lives <= 0) {
                    endGame("Mission Failed", "You ran out of lives! Review the facts and try again.");
                }
                // Question repeats automatically since currentQuestionIndex is unchanged
            }
        }

        // Keyboard controls for player ship
        window.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowLeft' || e.key === 'a') player.dx = -player.speed;
            if (e.key === 'ArrowRight' || e.key === 'd') player.dx = player.speed;
            if (e.key === ' ' || e.key === 'ArrowUp') {
                shootBullet();
                e.preventDefault();
            }
        });

        window.addEventListener('keyup', (e) => {
            if ((e.key === 'ArrowLeft' || e.key === 'a') && player.dx < 0) player.dx = 0;
            if ((e.key === 'ArrowRight' || e.key === 'd') && player.dx > 0) player.dx = 0;
        });

            if (player.x < 0) player.x = 0;
            if (player.x > canvas.width - player.width) player.x = canvas.width - player.width;

            // Check if clicked directly on an alien pod
            for (let j = aliens.length - 1; j >= 0; j--) {
                let a = aliens[j];
                if (clickX >= a.x && clickX <= a.x + a.width && clickY >= a.y && clickY <= a.y + a.height) {
                    createExplosion(a.x + a.width / 2, a.y + a.height / 2, a.isCorrect ? '#10b981' : '#f43f5e');
                    selectAnswer(a.text, shuffledQuestions[currentQuestionIndex].correctAnswer);
                    return;
                }
            }

            // Otherwise shoot laser bullet
            shootBullet();
        });

        function shootBullet() {
            if (!gameRunning) return;
            bullets.push({
                x: player.x + player.width / 2 - 3,
                y: player.y,
                width: 6,
                height: 12,
                speed: 8
            });
        }

        function update() {
            if (!gameRunning) return;

            // Move player
            player.x += player.dx;
            if (player.x < 0) player.x = 0;
            if (player.x > canvas.width - player.width) player.x = canvas.width - player.width;

            // Move bullets
            for (let i = bullets.length - 1; i >= 0; i--) {
                bullets[i].y -= bullets[i].speed;
                if (bullets[i].y < 0) {
                    bullets.splice(i, 1);
                    continue;
                }

                // Check collision with aliens
                let bulletHit = false;
                for (let j = aliens.length - 1; j >= 0; j--) {
                    let a = aliens[j];
                    if (
                        bullets[i] &&
                        bullets[i].x < a.x + a.width &&
                        bullets[i].x + bullets[i].width > a.x &&
                        bullets[i].y < a.y + a.height &&
                        bullets[i].y + bullets[i].height > a.y
                    ) {
                        // Collision detected!
                        bulletHit = true;
                        createExplosion(a.x + a.width / 2, a.y + a.height / 2, a.isCorrect ? '#10b981' : '#f43f5e');
                        selectAnswer(a.text, shuffledQuestions[currentQuestionIndex].correctAnswer);
                        break;
                    }
                }
                if (bulletHit) {
                    bullets.splice(i, 1);
                }
            }

            // Move aliens horizontally
            aliens.forEach(a => {
                a.x += a.speedX;
                if (a.x <= 10 || a.x + a.width >= canvas.width - 10) {
                    a.speedX *= -1;
                }
            });

            // Update particles
            for (let i = particles.length - 1; i >= 0; i--) {
                let p = particles[i];
                p.x += p.vx;
                p.y += p.vy;
                p.alpha -= 0.03;
                if (p.alpha <= 0) particles.splice(i, 1);
            }
        }

        function draw() {

            for (let i = 0; i < 40; i++) {
                let sx = (i * 37) % canvas.width;
                let sy = (i * 53 + Date.now() * 0.02) % canvas.height;
                ctx.globalAlpha = (i % 3) * 0.3 + 0.2;
                ctx.fillRect(sx, sy, 2, 2);
            }
            ctx.globalAlpha = 1.0;

            // Draw Player Ship
            ctx.fillStyle = '#38bdf8';
            ctx.beginPath();
            ctx.moveTo(player.x + player.width / 2, player.y);
            ctx.lineTo(player.x, player.y + player.height);
            ctx.lineTo(player.x + player.width, player.y + player.height);
            ctx.closePath();
            ctx.fill();

            // Draw Bullets
            ctx.fillStyle = '#facc15';
            bullets.forEach(b => {
                ctx.fillRect(b.x, b.y, b.width, b.height);
            });

            // Draw Aliens (Trivia Pods)
            aliens.forEach(a => {
                ctx.fillStyle = '#1e293b';
                ctx.strokeStyle = a.isCorrect ? '#38bdf8' : '#6366f1';
                ctx.lineWidth = 2;
                ctx.beginPath();
                ctx.roundRect(a.x, a.y, a.width, a.height, 10);
                ctx.fill();
                ctx.stroke();

                // Alien saucer dome
                ctx.fillStyle = '#06b6d4';
                ctx.beginPath();
                ctx.arc(a.x + a.width / 2, a.y + 12, 8, Math.PI, 0, false);
                ctx.fill();

                // Text snippet on alien pod
                ctx.fillStyle = '#f8fafc';
                ctx.font = '11px Inter, sans-serif';
                ctx.textAlign = 'center';
                let shortText = a.text.length > 18 ? a.text.substring(0, 16) + '...' : a.text;
                ctx.fillText(shortText, a.x + a.width / 2, a.y + 36);
            });

            // Draw Particles
            particles.forEach(p => {
                ctx.save();
                ctx.globalAlpha = p.alpha;
                ctx.fillStyle = p.color;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            });
        }

        function gameLoop() {
            if (!gameRunning) return;
            update();
            draw();
            requestAnimationFrame(gameLoop);
        }

        function endGame(title, message) {
            gameRunning = false;
            if (timerInterval) clearInterval(timerInterval);
            document.getElementById('gameOverTitle').innerText = title;
            document.getElementById('gameOverMessage').innerText = message;
            document.getElementById('finalScore').innerText = score;
            document.getElementById('gameOverScreen').classList.remove('hidden');
        }

        function resetGame() {
            startGame();
        }
    </script>
